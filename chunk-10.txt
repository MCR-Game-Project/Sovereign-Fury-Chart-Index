  cv.addEventListener("pointerdown", function(ev){
    panning=true; panMoved=false; lastPan={x:ev.clientX,y:ev.clientY}; cv.classList.add("panning");
    cv.setPointerCapture(ev.pointerId);
  });
  cv.addEventListener("pointermove", function(ev){
    var r=cv.getBoundingClientRect(), x=ev.clientX-r.left, y=ev.clientY-r.top;
    if(panning && lastPan){
      var dx=ev.clientX-lastPan.x, dy=ev.clientY-lastPan.y;
      if(Math.abs(dx)+Math.abs(dy)>3) panMoved=true;
      cam.x-=dx/cam.scale;
      cam.y+=dy/cam.scale;
      lastPan={x:ev.clientX,y:ev.clientY};
      draw();
    } else {
      hover=hitSheet(x,y);
      setStatus(hover);
      draw();
    }
  });
  function endPan(ev){
    if(panning && ev.type==="pointerup" && !panMoved){
      var r=cv.getBoundingClientRect(), s=hitSheet(ev.clientX-r.left, ev.clientY-r.top);
      if(s){ fitSheets([s], 0.35); setStatus(s); }
    }
    panning=false; lastPan=null; panMoved=false; cv.classList.remove("panning");
  }
  cv.addEventListener("pointerup", endPan);
  cv.addEventListener("pointercancel", endPan);
  cv.addEventListener("wheel", function(ev){
    ev.preventDefault();
    var r=cv.getBoundingClientRect();
    zoomAt(ev.clientX-r.left, ev.clientY-r.top, ev.deltaY<0?1.18:1/1.18);
  }, {passive:false});
