  function drawResChip(x, y, kind){
    var spec=RES_CHIP[kind]||RES_CHIP["iron-ore"], s=ICON, m=2.2;
    ctx.save();
    ctx.fillStyle="#f4ead0";
    ctx.strokeStyle=spec.border;
    ctx.lineWidth=1.4;
    roundRect(x+0.5, y+0.5, s-1, s-1, 2.2);
    ctx.fill(); ctx.stroke();
    ctx.fillStyle=spec.ink;
    if(kind==="charcoal"){
      ctx.fillRect(x+m, y+s*0.38, s-2*m, s*0.16);
      ctx.fillRect(x+m+0.6, y+s*0.56, s-2*m-1.2, s*0.15);
      ctx.fillRect(x+m+1.2, y+s*0.22, s-2*m-2.4, s*0.14);
    } else {
      ctx.beginPath(); ctx.arc(x+s*0.34, y+s*0.58, s*0.13, 0, Math.PI*2); ctx.fill();
      ctx.beginPath(); ctx.arc(x+s*0.56, y+s*0.40, s*0.14, 0, Math.PI*2); ctx.fill();
      ctx.beginPath(); ctx.arc(x+s*0.72, y+s*0.60, s*0.12, 0, Math.PI*2); ctx.fill();
    }
    ctx.restore();
  }
  function drawStationHex(ps){
    var rad=Math.max(18, HEX_MI*cam.scale*0.52);

    ctx.lineWidth=0.9;

  }
  function drawStations(){
    var i, st, pw, ps, res, n, rowW, x0, y0, k, nameX;
    if(cam.scale<6) return;
    ctx.save();
    for(i=0;i<STATIONS.length;i++){
      st=STATIONS[i];
      pw=lonlatToWorld(st.lon, st.lat);
      ps=worldToScreen(pw.x, pw.y);
      if(ps.x<-80||ps.x>stage.clientWidth+80||ps.y<-40||ps.y>stage.clientHeight+40) continue;
      drawStationHex(ps);
      res=st.res||[];
      n=res.length;
      rowW=n? n*ICON+(n-1)*ICON_GAP : 0;
      ctx.font='italic 13px "Iowan Old Style","Palatino Linotype",Palatino,Georgia,serif';
      ctx.textAlign="left"; ctx.textBaseline="middle";
      /* Name starts at the last drawn chip's right edge — never after a reserved empty slot. */
      nameX=n? (n-1)*(ICON+ICON_GAP)+ICON+NAME_GAP : 0;
      x0=ps.x-(nameX+ctx.measureText(st.name).width)/2;
      y0=ps.y-ICON/2;
      for(k=0;k<n;k++) drawResChip(x0+k*(ICON+ICON_GAP), y0, res[k]);
      ctx.lineWidth=3;
      ctx.strokeStyle="rgba(236,225,198,.92)";
      ctx.strokeText(st.name, x0+nameX, ps.y);
      ctx.fillStyle="#2c2214";
      ctx.fillText(st.name, x0+nameX, ps.y);
    }
    ctx.restore();
  }
  function hexPoly(cx, cy, r){
    var i, a;
    ctx.beginPath();
    for(i=0;i<6;i++){
      a=Math.PI/180*(60*i-30);
      if(i===0) ctx.moveTo(cx+r*Math.cos(a), cy+r*Math.sin(a));
      else ctx.lineTo(cx+r*Math.cos(a), cy+r*Math.sin(a));
    }
    ctx.closePath();
  }
  function drawHexes(s){
    var b=sheetWorld(s), pxPerHex=HEX_MI*cam.scale;
    if(pxPerHex<3) return;
    var origin=lonlatToWorld(s.box.w, s.box.s);
    var view=screenToWorld(0,0), view2=screenToWorld(stage.clientWidth, stage.clientHeight);
    var minX=Math.min(view.x,view2.x), maxX=Math.max(view.x,view2.x);
    var minY=Math.min(view.y,view2.y), maxY=Math.max(view.y,view2.y);
    var c0=Math.max(0, Math.floor((minX-origin.x)/HEX_MI)-2);
    var c1=Math.min(s.hex.cols-1, Math.ceil((maxX-origin.x)/HEX_MI)+2);
    var r0=Math.max(0, Math.floor((minY-origin.y)/ROW_MI)-2);
    var r1=Math.min(s.hex.rows-1, Math.ceil((maxY-origin.y)/ROW_MI)+2);
    var c,r,wx,wy,sp, rad=pxPerHex*0.52;
    ctx.strokeStyle="rgba(44,34,20,.28)";
    ctx.lineWidth=0.6;
    for(r=r0;r<=r1;r++){
      for(c=c0;c<=c1;c++){
        wx=origin.x+(c+((r&1)?0.5:0))*HEX_MI;
        wy=origin.y+r*ROW_MI;
        if(wx<b.x0||wx>b.x1||wy<b.y0||wy>b.y1) continue;
        sp=worldToScreen(wx,wy);
        hexPoly(sp.x, sp.y, rad);
        ctx.stroke();
      }
    }
  }
  function niceMiles(pxTarget){
    var mi=pxTarget/cam.scale, step=0.1, steps=[0.1,0.2,0.5,1,2,5,10,20,50,100,200,500], i;
    for(i=0;i<steps.length;i++) if(steps[i]>=mi){ step=steps[i]; break; }
    return step;
  }
