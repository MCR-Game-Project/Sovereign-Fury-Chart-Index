townThumb(){
    if(ytThumb) return ytThumb;
    var c=document.createElement("canvas"), t=c.getContext("2d");
    var W=400, H=400, col, row, k, fill, R=W/(1.5*(YT_COLS-1)+2), HH=R*SQRT3;
    function hx(col){ return R+1.5*R*col; }
    function hy(col,row){ return HH/2+HH*row+((col&1)?HH/2:0); }
    c.width=W; c.height=H;
    t.fillStyle="#e7ddc0"; t.fillRect(0,0,W,H);
    for(row=0;row<YT_ROWS;row++){
      for(col=0;col<YT_COLS;col++){
        k=col+","+row;
        if(ytCell(col,row)==="W") fill="#a8c0e8";
        else if(YT_TOWN_S[k]||YT_PORT_S[k]) fill="#8a6a48";
        else if(YT_HEAVY_S[k]) fill="#4a5738";
        else if(YT_LIGHT_S[k]) fill="#6f8158";
        else if(YT_CORN_S[k]) fill="#d9c07a";
        else if(YT_BEAN_S[k]) fill="#d5c48c";
        else if(ytCell(col,row)>=40) fill="#d4b57c";
        else if(ytCell(col,row)>=20) fill="#ead9b4";
        else fill="#ece2c8";
        t.beginPath();
        for(k=0;k<6;k++){
          var a=Math.PI/180*60*k, px=hx(col)+R*Math.cos(a), py=hy(col,row)+R*Math.sin(a);
          if(k===0) t.moveTo(px,py); else t.lineTo(px,py);
        }
        t.closePath(); t.fillStyle=fill; t.fill();
      }
    }
    ytThumb=c;
    return c;
  }
