  function plateUrls(file){
    var name=file.split("/").pop();
    return [
      file,
      "https://cdn.jsdelivr.net/gh/MCR-Game-Project/Sovereign-Fury-Chart-Index@main/"+name+"?v=yt3",
      "https://raw.githubusercontent.com/MCR-Game-Project/Sovereign-Fury-Chart-Index/main/"+name+"?v=yt3"
    ];
  }
  function loadSheetImage(s){
    if(!s.img) return;
    var urls=plateUrls(s.img), i=0, img;
    function next(){
      if(i>=urls.length) return;
      img=new Image();
      img.onload=function(){ s.plateImg=img; s.plateReady=true; if(cv && cv.width) draw(); };
      img.onerror=function(){ i++; next(); };
      img.src=urls[i];
    }
    next();
  }

  var ytThumb=null;
  function ytFill(col, row){
    var k=col+","+row;
    if(ytCell(col,row)==="W") return "#7ea0c4";
    if(YT_TOWN_S[k]||YT_PORT_S[k]) return "#8a6a48";
    if(YT_HEAVY_S[k] && ytOnLand([col,row])) return "#4a5738";
    if(YT_LIGHT_S[k] && ytOnLand([col,row])) return "#6f8158";
    if(YT_CORN_S[k] && ytOnLand([col,row])) return "#d9c07a";
    if(YT_BEAN_S[k] && ytOnLand([col,row])) return "#d5c48c";
    if(ytCell(col,row)>=40) return "#d4b57c";
    if(ytCell(col,row)>=20) return "#ead9b4";
    return "#ece2c8";
  }
  function yorktownThumb(){
    if(ytThumb) return ytThumb;
    var c=document.createElement("canvas"), t=c.getContext("2d");
    var W=400, H=400, col, row, k, R=W/(1.5*(YT_COLS-1)+2), HH=R*SQRT3;
    function hx(col){ return R+1.5*R*col; }
    function hy(col,row){ return HH/2+HH*row+((col&1)?HH/2:0); }
    c.width=W; c.height=H;
    t.fillStyle="#7ea0c4"; t.fillRect(0,0,W,H);
    for(row=0;row<YT_ROWS;row++){
      for(col=0;col<YT_COLS;col++){
        t.beginPath();
        for(k=0;k<6;k++){
          var a=Math.PI/180*60*k, px=hx(col)+R*Math.cos(a), py=hy(col,row)+R*Math.sin(a);
          if(k===0) t.moveTo(px,py); else t.lineTo(px,py);
        }
        t.closePath(); t.fillStyle=ytFill(col,row); t.fill();
      }
    }
    ytThumb=c;
    return c;
  }
  function ytHexCenter(s, c, r){
    var b=sheetWorld(s), Rmi=(b.x1-b.x0)/(1.5*(YT_COLS-1)+2), HH=Rmi*SQRT3;
    return {x:b.x0+Rmi+1.5*Rmi*c, y:b.y1-(HH/2+HH*r+((c&1)?HH/2:0)), R:Rmi};
  }
  function ytHexPath(s, c, r){
    var h=ytHexCenter(s,c,r), i, a, p;
    ctx.beginPath();
    for(i=0;i<6;i++){
      a=Math.PI/180*60*i;
      p=worldToScreen(h.x+h.R*Math.cos(a), h.y-h.R*Math.sin(a));
      if(i===0) ctx.moveTo(p.x,p.y); else ctx.lineTo(p.x,p.y);
    }
    ctx.closePath();
    return h;
  }
  function clipYorktownLand(s){
    var b=sheetWorld(s), nw=worldToScreen(b.x0,b.y1), se=worldToScreen(b.x1,b.y0);
    var i, p, q;
    ctx.beginPath();
    ctx.rect(nw.x, nw.y, se.x-nw.x, se.y-nw.y);
    for(i=0;i<YORK_ESTUARY.length;i++){
      p=lonlatToWorld(YORK_ESTUARY[i][0], YORK_ESTUARY[i][1]);
      q=worldToScreen(p.x, p.y);
      if(i===0) ctx.moveTo(q.x,q.y); else ctx.lineTo(q.x,q.y);
    }
    ctx.closePath();
    ctx.clip("evenodd");
  }
