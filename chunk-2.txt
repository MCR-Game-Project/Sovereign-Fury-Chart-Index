b.x0+Rmi+1.5*Rmi*c,
      y:b.y1-(HH/2+HH*r+((c&1)?HH/2:0)),
      R:Rmi
    };
  }
  function ytHexPolyWorld(cx, cy, R){
    var i, a, p, first;
    ctx.beginPath();
    for(i=0;i<6;i++){
      a=Math.PI/180*60*i;
      p=worldToScreen(cx+R*Math.cos(a), cy-R*Math.sin(a));
      if(i===0){ ctx.moveTo(p.x,p.y); first=p; }
      else ctx.lineTo(p.x,p.y);
    }
    ctx.closePath();
  }
  function drawYorktownHexes(s){
    var b=sheetWorld(s), c, r, h, k, fill, p, road;
    var px=HEX_MI*cam.scale;
    if(px<2) return;
    for(r=0;r<YT_ROWS;r++){
      for(c=0;c<YT_COLS;c++){
        h=ytHexCenter(b,c,r); k=c+","+r;
        if(ytCell(c,r)==="W") fill="#a8c0e8";
        else if(YT_TOWN_S[k]||YT_PORT_S[k]) fill="#8a6a48";
        else if(YT_HEAVY_S[k]) fill="#4a5738";
        else if(YT_LIGHT_S[k]) fill="#6f8158";
        else if(YT_CORN_S[k]) fill="#d9c07a";
        else if(YT_BEAN_S[k]) fill="#d5c48c";
        else if(ytCell(c,r)>=40) fill="#d4b57c";
        else if(ytCell(c,r)>=20) fill="#ead9b4";
        else fill="#ece2c8";
        ytHexPolyWorld(h.x,h.y,h.R);
        ctx.fillStyle=fill;
        ctx.fill();
        ctx.strokeStyle="rgba(90,74,50,.28)";
        ctx.lineWidth=0.55;
        ctx.stroke();
      }
    }
    ctx.strokeStyle="#6c4530";
    ctx.lineWidth=Math.max(1.1, px*0.12);
    ctx.lineCap="round"; ctx.lineJoin="round";
    ctx.beginPath();
    for(c=0;c<YT_ROAD.length;c++){
      p=ytHexCenter(b, YT_ROAD[c][0], YT_ROAD[c][1]);
      p=worldToScreen(p.x,p.y);
      if(c===0) ctx.moveTo(p.x,p.y); else ctx.lineTo(p.x,p.y);
    }
    ctx.stroke();
    /* Yorktown Creek: hexsides, inland south → mouth north into the York. */
    var creek=[[7,5,3],[7,5,2]], rr;
    for(rr=6;rr<=16;rr++){ creek.push([6,rr,0]); creek.push([7,rr,2]); }
    ctx.strokeStyle="#2e5c88";
    ctx.lineWidth=Math.max(1.4, px*0.16);
    ctx.beginPath();
    for(c=0;c<creek.length;c++){
      h=ytHexCenter(b, creek[c][0], creek[c][1]);
      var a=Math.PI/180*60*creek[c][2], a2=Math.PI/180*60*((creek[c][2]+1)%6);
      var mx=h.x+h.R*(Math.cos(a)+Math.cos(a2))/2;
      var my=h.y-h.R*(Math.sin(a)+Math.sin(a2))/2;
      p=worldToScreen(mx,my);
      if(c===0) ctx.moveTo(p.x,p.y); else ctx.lineTo(p.x,p.y);
    }
    ctx.stroke();
    if(px>=8){
      ctx.fillStyle="#2c2214";
      ctx.font='700 12px "Iowan Old Style","Palatino Linotype",Palatino,Georgia,serif';
      ctx.textAlign="center"; ctx.textBaseline="middle";
      p=ytHexCenter(b,10,6); p=worldToScreen(p.x,p.y);
      ctx.fillText("Yorktown", p.x, p.y);
      ctx.font='italic 11px "Iowan Old Style","Palatino Linotype",Palatino,Georgia,serif';
      ctx.fillStyle="#7a1f1f";
      p=ytHexCenter(b,8,2); p=worldToScreen(p.x,p.y);
      ctx.fillText("York River", p.x, p.y);
      ctx.fillStyle="#5a3410";
      p=ytHexCenter(b,14,0); p=worldToScreen(p.x,p.y);
      ctx.fillText("Gloucester Pt.", p.x, p.y+10);
      ctx.fillStyle="#2e5c88";
      p=ytHexCenter(b,6,9); p=worldToScreen(p.x,p.y);
      ctx.textAlign="left";
      ctx.fillText("Yorktown Cr.", p.x+6, p.y);
    }
  }
  function drawYorktownPlate(s, x, y, w, h){
    if(ytPlateReady && w>=12){
      ctx.drawImage(ytPlateImg, x, y, w, h);
      return true;
    }
    if(w>=16){ drawYorktownHexes(s); return true; }
    return false;
  }
