      if(i===0) ctx.moveTo(s.x,s.y); else ctx.lineTo(s.x,s.y);
    }

  }
  function lonlatInPoly(lon, lat, poly){
    var inside=false, n=poly.length, j=n-1, i, xi, yi, xj, yj;
    for(i=0;i<n;i++){
      xi=poly[i][0]; yi=poly[i][1]; xj=poly[j][0]; yj=poly[j][1];
      if(((yi>lat)!==(yj>lat)) && (lon<(xj-xi)*(lat-yi)/(yj-yi+1e-30)+xi)) inside=!inside;
      j=i;
    }
    return inside;
  }
  /* Theatre land is clipped at this latitude. River strokes stop on that edge. */
  var MAP_NORTH=43.4;
  function clipRiverPts(pts){
    var out=[], i, a, b, t, p;
    for(i=0;i<pts.length;i++){
      p=pts[i];
      if(p[1]<=MAP_NORTH){
        if(i && pts[i-1][1]>MAP_NORTH){
          a=pts[i-1]; b=p;
          t=(MAP_NORTH-a[1])/(b[1]-a[1]);
          out.push([a[0]+t*(b[0]-a[0]), MAP_NORTH]);
        }
        out.push(p);
      } else if(i && pts[i-1][1]<=MAP_NORTH){
        a=pts[i-1]; b=p;
        t=(MAP_NORTH-a[1])/(b[1]-a[1]);
        out.push([a[0]+t*(b[0]-a[0]), MAP_NORTH]);
      }
    }
    return out.length? out : pts;
  }
  /* River strokes stop at a larger body. A river may run through its own estuary. */
  var OPEN_WATER_NAMES={"Chesapeake Bay":1,"Hampton Roads":1,"Mobile Bay":1};
  function riverInLargerWater(name, lon, lat){
    var i, L, own=name+" estuary";
    for(i=0;i<LAKES.length;i++){
      L=LAKES[i];
      if(L.name===own) continue;
      if(OPEN_WATER_NAMES[L.name] || / estuary$/.test(L.name)){
        if(lonlatInPoly(lon, lat, L.pts)) return true;
      }
    }
    return false;
  }
  function drawRiver(name, pts){
    var i, p, s, drawing=false;
    pts=clipRiverPts(pts);
    ctx.beginPath();
    for(i=0;i<pts.length;i++){
      if(riverInLargerWater(name, pts[i][0], pts[i][1])){
        if(drawing){ ctx.stroke(); ctx.beginPath(); drawing=false; }
        continue;
      }
      p=lonlatToWorld(pts[i][0], pts[i][1]); s=worldToScreen(p.x,p.y);
      if(!drawing){ ctx.moveTo(s.x,s.y); drawing=true; }
      else ctx.lineTo(s.x,s.y);
    }
    if(drawing) ctx.stroke();
  }
  function roundRect(x, y, w, h, r){
    ctx.beginPath();
    ctx.moveTo(x+r, y);
    ctx.arcTo(x+w, y, x+w, y+h, r);
    ctx.arcTo(x+w, y+h, x, y+h, r);
    ctx.arcTo(x, y+h, x, y, r);
    ctx.arcTo(x, y, x+w, y, r);
    ctx.closePath();
  }

      ctx.fillRect(x+m, y+s*0.38, s-2*m, s*0.16);
      ctx.fillRect(x+m+0.6, y+s*0.56, s-2*m-1.2, s*0.15);
      ctx.fillRect(x+m+1.2, y+s*0.22, s-2*m-2.4, s*0.14);
    } else {
      ctx.beginPath(); ctx.arc(x+s*0.34, y+s*0.58, s*0.13, 0, Math.PI*2); ctx.fill();
      ctx.beginPath(); ctx.arc(x+s*0.56, y+s*0.40, s*0.14, 0, Math.PI*2); ctx.fill();
      ctx.beginPath(); ctx.arc(x+s*0.72, y+s*0.60, s*0.12, 0, Math.PI*2); ctx.fill();
    }
    ctx.restore();
  }
