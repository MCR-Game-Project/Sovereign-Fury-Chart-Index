  function drawRiver(name, pts){
    var i, p, s, drawing=false;
    ctx.beginPath();
    for(i=0;i<pts.length;i++){
      if(riverInLargerWater(name, pts[i][0], pts[i][1])){
        if(drawing){ ctx.stroke(); ctx.beginPath(); drawing=false; }
        continue;
      }
      p=lonlatToWorld(pts[i][0], pts[i][1]); s=worldToScreen(p.x,p.y);
      if(!drawing){ ctx.moveTo(s.x,s.y); drawing=true; }
      else ctx.lineTo(s.x,s.y);
    }
    if(drawing) ctx.stroke();
  }
  function hexPoly(cx, cy, r){
    var i, a;
    ctx.beginPath();
    for(i=0;i<6;i++){
      a=Math.PI/180*(60*i-30);
      if(i===0) ctx.moveTo(cx+r*Math.cos(a), cy+r*Math.sin(a));
      else ctx.lineTo(cx+r*Math.cos(a), cy+r*Math.sin(a));
    }
    ctx.closePath();
  }
  function drawHexes(s){
    var b=sheetWorld(s), pxPerHex=HEX_MI*cam.scale;
    if(pxPerHex<5) return;
    var origin=lonlatToWorld(s.box.w, s.box.s);
    var view=screenToWorld(0,0), view2=screenToWorld(stage.clientWidth, stage.clientHeight);
    var minX=Math.min(view.x,view2.x), maxX=Math.max(view.x,view2.x);
    var minY=Math.min(view.y,view2.y), maxY=Math.max(view.y,view2.y);
    var c0=Math.max(0, Math.floor((minX-origin.x)/HEX_MI)-2);
    var c1=Math.min(s.hex.cols-1, Math.ceil((maxX-origin.x)/HEX_MI)+2);
    var r0=Math.max(0, Math.floor((minY-origin.y)/ROW_MI)-2);
    var r1=Math.min(s.hex.rows-1, Math.ceil((maxY-origin.y)/ROW_MI)+2);
    var c,r,wx,wy,sp, rad=pxPerHex*0.52;
    ctx.strokeStyle="rgba(44,34,20,.28)";
    ctx.lineWidth=0.6;
    for(r=r0;r<=r1;r++){
      for(c=c0;c<=c1;c++){
        wx=origin.x+(c+((r&1)?0.5:0))*HEX_MI;
        wy=origin.y+r*ROW_MI;
        if(wx<b.x0||wx>b.x1||wy<b.y0||wy>b.y1) continue;
        sp=worldToScreen(wx,wy);
        hexPoly(sp.x, sp.y, rad);
        ctx.stroke();
      }
    }
  }

    var mi=pxTarget/cam.scale, step=0.1, steps=[0.1,0.2,0.5,1,2,5,10,20,50,100,200,500], i;
    for(i=0;i<steps.length;i++) if(steps[i]>=mi){ step=steps[i]; break; }
    return step;
  }
