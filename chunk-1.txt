  function pipLonlat(lon, lat, poly){
    var n=poly.length, inside=false, i, x1,y1,x2,y2,xint;
    for(i=0;i<n;i++){
      x1=poly[i][0]; y1=poly[i][1];
      x2=poly[(i+1)%n][0]; y2=poly[(i+1)%n][1];
      if((y1>lat)!==(y2>lat)){
        xint=x1+(lat-y1)*(x2-x1)/(y2-y1);
        if(lon<xint) inside=!inside;
      }
    }
    return inside;
  }
