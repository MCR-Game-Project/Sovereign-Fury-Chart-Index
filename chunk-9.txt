-cam.x)*cam.scale+stage.clientWidth/2, y:(cam.y-wy)*cam.scale+stage.clientHeight/2};
  }
  function screenToWorld(sx, sy){
    return {x:cam.x+(sx-stage.clientWidth/2)/cam.scale, y:cam.y-(sy-stage.clientHeight/2)/cam.scale};
  }
  function sheetWorld(s){
    var sw=lonlatToWorld(s.box.w, s.box.s), ne=lonlatToWorld(s.box.e, s.box.n);
    return {x0:sw.x, y0:sw.y, x1:ne.x, y1:ne.y};
  }
  function zoomAt(sx, sy, factor){
    var w=screenToWorld(sx,sy);
    cam.scale=Math.max(MIN_SCALE, Math.min(MAX_SCALE, cam.scale*factor));
    var w2=screenToWorld(sx,sy);
    cam.x+=w.x-w2.x; cam.y+=w.y-w2.y;
    draw();
  }
  function fitSheets(list, pad){
    var i, b, x0=1e9,y0=1e9,x1=-1e9,y1=-1e9;
    for(i=0;i<list.length;i++){
      b=sheetWorld(list[i]);
      if(b.x0<x0) x0=b.x0; if(b.y0<y0) y0=b.y0;
      if(b.x1>x1) x1=b.x1; if(b.y1>y1) y1=b.y1;
    }
    pad=pad==null?18:pad;
    x0-=pad; y0-=pad; x1+=pad; y1+=pad;
    var W=stage.clientWidth, H=stage.clientHeight;
    cam.x=(x0+x1)/2; cam.y=(y0+y1)/2;
    cam.scale=Math.max(MIN_SCALE, Math.min(MAX_SCALE, Math.min(W/(x1-x0), H/(y1-y0))*0.92));
    draw();
  }
  function hitSheet(sx, sy){
    var w=screenToWorld(sx,sy), i, b, c, d, best=null, bestD=16;
    for(i=SHEETS.length-1;i>=0;i--){
      b=sheetWorld(SHEETS[i]);
      if(w.x>=b.x0 && w.x<=b.x1 && w.y>=b.y0 && w.y<=b.y1) return SHEETS[i];
    }
    for(i=0;i<SHEETS.length;i++){
      b=sheetWorld(SHEETS[i]);
      c=worldToScreen((b.x0+b.x1)/2,(b.y0+b.y1)/2);
      d=Math.hypot(sx-c.x, sy-c.y);
      if(d<bestD){ bestD=d; best=SHEETS[i]; }
    }
    return best;
  }
  function drawPath(pts, close){
    var i, p, s;
    ctx.beginPath();
    for(i=0;i<pts.length;i++){
      p=lonlatToWorld(pts[i][0], pts[i][1]); s=worldToScreen(p.x,p.y);
      if(i===0) ctx.moveTo(s.x,s.y); else ctx.lineTo(s.x,s.y);
    }
    if(close) ctx.closePath();
  }
  function lonlatInPoly(lon, lat, poly){
    var inside=false, n=poly.length, j=n-1, i, xi, yi, xj, yj;
    for(i=0;i<n;i++){
      xi=poly[i][0]; yi=poly[i][1]; xj=poly[j][0]; yj=poly[j][1];
      if(((yi>lat)!==(yj>lat)) && (lon<(xj-xi)*(lat-yi)/(yj-yi+1e-30)+xi)) inside=!inside;
      j=i;
    }
    return inside;
  }
  /* Theatre land is clipped at this latitude. River strokes stop on that edge. */
  var MAP_NORTH=43.4;
  function clipRiverPts(pts){
    var out=[], i, a, b, t, p;
    for(i=0;i<pts.length;i++){
      p=pts[i];
      if(p[1]<=MAP_NORTH){
        if(i && pts[i-1][1]>MAP_NORTH){
          a=pts[i-1]; b=p;
          t=(MAP_NORTH-a[1])/(b[1]-a[1]);
          out.push([a[0]+t*(b[0]-a[0]), MAP_NORTH]);
        }
        out.push(p);
      } else if(i && pts[i-1][1]<=MAP_NORTH){
        a=pts[i-1]; b=p;
        t=(MAP_NORTH-a[1])/(b[1]-a[1]);
        out.push([a[0]+t*(b[0]-a[0]), MAP_NORTH]);
      }
    }
    return out.length? out : pts;
  }
  /* River strokes stop at a larger body. A river may run through its own estuary. */
  var OPEN_WATER_NAMES={"Chesapeake Bay":1,"Hampton Roads":1,"Mobile Bay":1};
  function riverInLargerWater(name, lon, lat){
    var i, L, own=name+" estuary";
    for(i=0;i<LAKES.length;i++){
      L=LAKES[i];
      if(L.name===own) continue;
      if(OPEN_WATER_NAMES[L.name] || / estuary$/.test(L.name)){
        if(lonlatInPoly(lon, lat, L.pts)) return true;
      }
    }
    return false;
  }
  function drawRiver(name, pts){
    var i, p, s, drawing=false;
    pts=clipRiverPts(pts);
    ctx.beginPath();
    for(i=0;i<pts.length;i++){
      if(riverInLargerWater(name, pts[i][0], pts[i][1])){
        if(drawing){ ctx.stroke(); ctx.beginPath(); drawing=false; }
        continue;
      }
      p=lonlatToWorld(pts[i][0], pts[i][1]); s=worldToScreen(p.x,p.y);
      if(!drawing){ ctx.moveTo(s.x,s.y); drawing=true; }
      else ctx.lineTo(s.x,s.y);
    }
    if(drawing) ctx.stroke();
  }
