  function worldToScreen(wx, wy){
    return {x:(wx-cam.x)*cam.scale+stage.clientWidth/2, y:(cam.y-wy)*cam.scale+stage.clientHeight/2};
  }
  function screenToWorld(sx, sy){
    return {x:cam.x+(sx-stage.clientWidth/2)/cam.scale, y:cam.y-(sy-stage.clientHeight/2)/cam.scale};
  }
  function sheetWorld(s){
    var sw=lonlatToWorld(s.box.w, s.box.s), ne=lonlatToWorld(s.box.e, s.box.n);
    return {x0:sw.x, y0:sw.y, x1:ne.x, y1:ne.y};
  }
  function geoScreenRect(s){
    var b=sheetWorld(s), a=worldToScreen(b.x0,b.y1), c=worldToScreen(b.x1,b.y0);
    return {x:Math.min(a.x,c.x), y:Math.min(a.y,c.y), w:Math.abs(c.x-a.x), h:Math.abs(c.y-a.y)};
  }
  function expandRect(r, minSide){
    if(r.w>=minSide && r.h>=minSide) return {x:r.x, y:r.y, w:r.w, h:r.h, grown:false};
    var aspect=r.w/Math.max(0.001,r.h), w=r.w, h=r.h;
    if(w<minSide){ w=minSide; h=w/aspect; }
    if(h<minSide){ h=minSide; w=h*aspect; }
    return {x:r.x+r.w/2-w/2, y:r.y+r.h/2-h/2, w:w, h:h, grown:true};
  }
  function unionRects(rs){
    var i, x0=1e9,y0=1e9,x1=-1e9,y1=-1e9;
    for(i=0;i<rs.length;i++){
      if(rs[i].x<x0) x0=rs[i].x;
      if(rs[i].y<y0) y0=rs[i].y;
      if(rs[i].x+rs[i].w>x1) x1=rs[i].x+rs[i].w;
      if(rs[i].y+rs[i].h>y1) y1=rs[i].y+rs[i].h;
    }
    return {x:x0, y:y0, w:x1-x0, h:y1-y0};
  }
  function sheetDisplayRect(s){
    var i, block=[], geos=[], u, e, g, sx, sy;
    if(PLATE_BLOCK.indexOf(s.id)>=0){
      for(i=0;i<SHEETS.length;i++) if(PLATE_BLOCK.indexOf(SHEETS[i].id)>=0){
        block.push(SHEETS[i]);
        geos.push(geoScreenRect(SHEETS[i]));
      }
      u=unionRects(geos);
      if(u.w<MIN_BLOCK || u.h<MIN_BLOCK){
        e=expandRect(u, MIN_BLOCK);
        sx=e.w/Math.max(0.001,u.w); sy=e.h/Math.max(0.001,u.h);
        for(i=0;i<block.length;i++) if(block[i]===s){
          g=geos[i];
          return {x:e.x+(g.x-u.x)*sx, y:e.y+(g.y-u.y)*sy, w:g.w*sx, h:g.h*sy, grown:true};
        }
      }
    }
    return expandRect(geoScreenRect(s), MIN_PLATE);
  }
  function zoomAt(sx, sy, factor){
    var w=screenToWorld(sx,sy);
    cam.scale=Math.max(MIN_SCALE, Math.min(MAX_SCALE, cam.scale*factor));
    var w2=screenToWorld(sx,sy);
    cam.x+=w.x-w2.x; cam.y+=w.y-w2.y;
    draw();
  }
  function fitSheets(list, pad){
    var i, b, x0=1e9,y0=1e9,x1=-1e9,y1=-1e9;
    for(i=0;i<list.length;i++){
      b=sheetWorld(list[i]);
      if(b.x0<x0) x0=b.x0; if(b.y0<y0) y0=b.y0;
      if(b.x1>x1) x1=b.x1; if(b.y1>y1) y1=b.y1;
    }
    pad=pad==null?18:pad;
    x0-=pad; y0-=pad; x1+=pad; y1+=pad;
    var W=stage.clientWidth, H=stage.clientHeight;
    cam.x=(x0+x1)/2; cam.y=(y0+y1)/2;
    cam.scale=Math.max(MIN_SCALE, Math.min(MAX_SCALE, Math.min(W/(x1-x0), H/(y1-y0))*0.92));
    draw();
  }
  function hitSheet(sx, sy){
    var i, r, d, best=null, bestD=1e9;
    for(i=SHEETS.length-1;i>=0;i--){
      r=sheetDisplayRect(SHEETS[i]);
      if(sx>=r.x && sx<=r.x+r.w && sy>=r.y && sy<=r.y+r.h){
        d=Math.hypot(sx-(r.x+r.w/2), sy-(r.y+r.h/2));
        if(d<bestD){ bestD=d; best=SHEETS[i]; }
      }
    }
    if(best) return best;
    for(i=0;i<SHEETS.length;i++){
      r=sheetDisplayRect(SHEETS[i]);
      d=Math.hypot(sx-(r.x+r.w/2), sy-(r.y+r.h/2));
      if(d<20 && d<bestD){ bestD=d; best=SHEETS[i]; }
    }
    return best;
  }
